delay.c



```c
void delay(uint timer)
{
    uchar j = 124;
    while(timer--)
    {
        while(j--);
    }
}
```

```c
void display(uint k)
{
    uchar i;
    for(i=0; i<segN; i++)
    {
        int result=1;
        num[segN-i-1]=k/result%10;
        result*=10;
        k/=10;
        P2 = segW[i];
        P0 = segD[num[segN-i-1]];
        delay(1);
    }   
}
```
